> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A pip-installable Reticulum interface plugin (`rns-push`) that keeps a registered phone's TCP interface alive after its socket dies, turns what would have been written to it into APNs pushes, and routes the phone's proofs back to the sender.

**Architecture:** `PushTCPServerInterface` subclasses RNS's `TCPServerInterface` and spawns `PushClientInterface`s that go dormant instead of detaching when bound to a registration. A `RegistrationService` exposes `rnspush.registration` with `/register` and `/unregister` link-request handlers, a `Registry` persists registrations, and a `PushDispatcher` batches outgoing packets into pushes sent by a small synchronous HTTP/2 `APNsClient`. A one-line shim in `<configdir>/interfaces/` makes stock `rnsd` load it.

**Tech Stack:** Python ≥3.8, `rns` 1.5.x (PyPI, unmodified), `h2` 4.x, `cryptography` (via `rns`), stdlib `unittest`.

## Review Focus

1. A client sends `token` as a hex **string** instead of msgpack bin (an easy client bug): `/register` must answer `bad_request`, not store a broken registration. Test in Task 6.
2. `allowed_bundle_ids` written as a single value, a comma list with spaces, or a ConfigObj list: all must parse to clean ids. Test in Task 7.
3. A bound interface torn down twice (a send failure then the read loop's own error): it must stay dormant and attached, not be removed on the second call. Test in Task 5.
4. A phone reconnects while its old connection is still live (half-open): binding the new interface must unbind the old one without retiring it, so the old one later tears down the stock way. Tests in Tasks 5 and 7.
5. An announce for a registered hash whose identity key differs from the registered key (a different identity colliding, or a corrupted entry) must not bind. Test in Task 6.

```python
from cryptography.hazmat.primitives.asymmetric import ec

TESTS_DIR = os.path.dirname(os.path.abspath(__file__))


def ensure_reticulum():
    """One RNS instance per test process. Interface.__init__ needs it."""
    instance = RNS.Reticulum.get_instance()
    if instance is None:
        instance = RNS.Reticulum(configdir=os.path.join(TESTS_DIR, "rnsconfig"), loglevel=RNS.LOG_ERROR)
    return instance


def make_raw(packet_type, destination_hash, header_type=RNS.Packet.HEADER_1, ifac=False,
             transport_id=b"\x11" * 16, context=0, payload=b"payload"):
    flags = (header_type << 6) | packet_type
    if header_type == RNS.Packet.HEADER_2:
        flags |= RNS.Transport.TRANSPORT << 4
    if ifac:
        flags |= 0x80
    raw = bytes([flags, 0])
    if header_type == RNS.Packet.HEADER_2:
        raw += transport_id
    return raw + destination_hash + bytes([context]) + payload


def tcp_pair():
    listener = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    listener.bind(("127.0.0.1", 0))
    listener.listen(1)
    client = socket.create_connection(listener.getsockname())
    server_side, _ = listener.accept()
    listener.close()
    return client, server_side


def p256_pem(curve=None):
    key = ec.generate_private_key(curve or ec.SECP256R1())
    pem = key.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8,
                            serialization.NoEncryption())
    return key, pem
```

```python
import collections

import RNS

DST_LEN = RNS.Reticulum.TRUNCATED_HASHLENGTH // 8

Header = collections.namedtuple("Header", "ifac header_type packet_type destination_hash")


def parse_header(raw):
    """Read the unmasked Reticulum header of a raw packet. None when too short."""
    if not raw:
        return None
    flags = raw[0]
    header_type = (flags & 0b01000000) >> 6
    offset = 2 + DST_LEN if header_type == RNS.Packet.HEADER_2 else 2
    if len(raw) < offset + DST_LEN + 1:
        return None
    return Header(ifac=bool(flags & 0x80),
                  header_type=header_type,
                  packet_type=flags & 0b00000011,
                  destination_hash=bytes(raw[offset:offset + DST_LEN]))
```

`rns_push/registry.py`:
```python
import dataclasses
import json
import os
import threading
import time

import RNS

REGISTRATION_LIFETIME = 30 * 24 * 60 * 60
ENVIRONMENTS = ("production", "sandbox")


@dataclasses.dataclass(frozen=True)
class Registration:
    destination_hash: bytes
    identity_public_key: bytes
    token: bytes
    environment: str
    bundle_id: str
    registered_at: float

    @property
    def expires_at(self):
        return self.registered_at + REGISTRATION_LIFETIME

    def to_json(self):
        return {"destination_hash": self.destination_hash.hex(),
                "identity_public_key": self.identity_public_key.hex(),
                "token": self.token.hex(),
                "environment": self.environment,
                "bundle_id": self.bundle_id,
                "registered_at": self.registered_at}

    @classmethod
    def from_json(cls, item):
        return cls(destination_hash=bytes.fromhex(item["destination_hash"]),
                   identity_public_key=bytes.fromhex(item["identity_public_key"]),
                   token=bytes.fromhex(item["token"]),
                   environment=str(item["environment"]),
                   bundle_id=str(item["bundle_id"]),
                   registered_at=float(item["registered_at"]))


def delivery_hash(identity):
    return RNS.Destination.hash(identity, "lxmf", "delivery")


class Registry:
    def __init__(self, path, clock=time.time):
        self.path = path
        self.clock = clock
        self._lock = threading.Lock()
        self._entries = {}
        self._load()

    def put(self, registration):
        with self._lock:
            self._entries[registration.destination_hash] = registration
            self._persist()

    def get(self, destination_hash):
        with self._lock:
            return self._entries.get(destination_hash)

    def remove(self, destination_hash):
        with self._lock:
            existed = self._entries.pop(destination_hash, None) is not None
            if existed:
                self._persist()
            return existed

    def all(self):
        with self._lock:
            return list(self._entries.values())

    def expire(self):
        now = self.clock()
        with self._lock:
            expired = [h for h, r in self._entries.items() if now >= r.expires_at]
            for h in expired:
                del self._entries[h]
            if expired:
                self._persist()
            return expired

    def _load(self):
        if not os.path.isfile(self.path):
            return
        try:
            with open(self.path) as f:
                data = json.load(f)
            entries = {}
            for item in data["registrations"]:
                registration = Registration.from_json(item)
                entries[registration.destination_hash] = registration
            self._entries = entries
        except Exception as e:
            corrupt = f"{self.path}.corrupt-{int(self.clock())}"
            RNS.log(f"Push registry at {self.path} is unreadable ({e}); moving it to {corrupt} and starting empty", RNS.LOG_ERROR)
            try:
                os.replace(self.path, corrupt)
            except OSError as move_error:
                RNS.log(f"Could not move the unreadable push registry aside: {move_error}", RNS.LOG_ERROR)

    def _persist(self):
        tmp = self.path + ".tmp"
        try:
            os.makedirs(os.path.dirname(self.path) or ".", exist_ok=True)
            with open(tmp, "w") as f:
                json.dump({"v": 1, "registrations": [r.to_json() for r in self._entries.values()]}, f)
            os.replace(tmp, self.path)
        except Exception as e:
            RNS.log(f"Could not write the push registry to {self.path}: {e}", RNS.LOG_ERROR)
```

```python
import json
import socket
import threading

import h2.config
import h2.connection
import h2.events


class FakeH2Server:
    """In-process HTTP/2 server. connect(host) hands APNsClient the client end of a fresh socket pair."""

    def __init__(self):
        self.requests = []
        self.hosts = []
        self.responses = []
        self.drop_next = False
        self._lock = threading.Lock()

    @property
    def connections(self):
        return len(self.hosts)

    def connect(self, host):
        client, server = socket.socketpair()
        client.settimeout(5)
        self.hosts.append(host)
        threading.Thread(target=self._serve, args=(server,), daemon=True).start()
        return client

    def _next_response(self):
        with self._lock:
            return self.responses.pop(0) if self.responses else (200, None)

    def _serve(self, sock):
        conn = h2.connection.H2Connection(config=h2.config.H2Configuration(client_side=False, header_encoding="utf-8"))
        conn.initiate_connection()
        headers, bodies = {}, {}
        try:
            sock.sendall(conn.data_to_send())
            while True:
                data = sock.recv(65535)
                if not data:
                    return
                for event in conn.receive_data(data):
                    if isinstance(event, h2.events.RequestReceived):
                        headers[event.stream_id] = dict(event.headers)
                        bodies[event.stream_id] = b""
                    elif isinstance(event, h2.events.DataReceived):
                        bodies[event.stream_id] += event.data
                        conn.acknowledge_received_data(event.flow_controlled_length, event.stream_id)
                    elif isinstance(event, h2.events.StreamEnded):
                        with self._lock:
                            self.requests.append((headers[event.stream_id], bodies[event.stream_id]))
                        if self.drop_next:
                            self.drop_next = False
                            return
                        status, reason = self._next_response()
                        body = json.dumps({"reason": reason}).encode() if reason else b""
                        conn.send_headers(event.stream_id, [(":status", str(status))], end_stream=not body)
                        if body:
                            conn.send_data(event.stream_id, body, end_stream=True)
                out = conn.data_to_send()
                if out:
                    sock.sendall(out)
        except OSError:
            return
        finally:
            sock.close()
```

```python
import base64
import json
import socket
import ssl
import time

import h2.config
import h2.connection
import h2.events
import RNS
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.primitives.asymmetric.utils import decode_dss_signature

HOSTS = {"production": "api.push.apple.com", "sandbox": "api.sandbox.push.apple.com"}
CONNECTION_FAILED = (0, "connection")
JWT_LIFETIME = 50 * 60
IDLE_REOPEN = 30 * 60


def _b64url(data):
    return base64.urlsafe_b64encode(data).rstrip(b"=")


def _compact_json(value):
    return json.dumps(value, separators=(",", ":")).encode("utf-8")


class JWTSigner:
    """ES256 provider token for APNs token-based auth, cached for 50 minutes."""

    def __init__(self, key_pem, key_id, team_id, clock=time.time):
        try:
            key = serialization.load_pem_private_key(key_pem, password=None)
        except Exception as e:
            raise ValueError(f"APNs key is not a readable PEM private key: {e}")
        if not isinstance(key, ec.EllipticCurvePrivateKey) or key.curve.name != "secp256r1":
            raise ValueError("APNs key must be a P-256 (ES256) private key")
        self._key = key
        self.key_id = key_id
        self.team_id = team_id
        self.clock = clock
        self._token = None
        self._issued_at = None

    def token(self):
        now = self.clock()
        if self._token is None or now - self._issued_at >= JWT_LIFETIME:
            signing_input = (_b64url(_compact_json({"alg": "ES256", "kid": self.key_id})) + b"." +
                             _b64url(_compact_json({"iss": self.team_id, "iat": int(now)})))
            r, s = decode_dss_signature(self._key.sign(signing_input, ec.ECDSA(hashes.SHA256())))
            signature = r.to_bytes(32, "big") + s.to_bytes(32, "big")
            self._token = (signing_input + b"." + _b64url(signature)).decode("ascii")
            self._issued_at = now
        return self._token

    def invalidate(self):
        self._token = None


def tls_connect(host, port=443, timeout=10):
    context = ssl.create_default_context()
    context.set_alpn_protocols(["h2"])
    raw = socket.create_connection((host, port), timeout=timeout)
    tls = context.wrap_socket(raw, server_hostname=host)
    if tls.selected_alpn_protocol() != "h2":
        tls.close()
        raise ConnectionError(f"{host} did not negotiate HTTP/2")
    return tls


class _Connection:
    def __init__(self, sock, now):
        self.sock = sock
        self.closed = False
        self.last_used = now
        self.h2 = h2.connection.H2Connection(config=h2.config.H2Configuration(client_side=True, header_encoding="utf-8"))
        self.h2.initiate_connection()
        self.sock.sendall(self.h2.data_to_send())

    def request(self, headers, body):
        stream_id = self.h2.get_next_available_stream_id()
        self.h2.send_headers(stream_id, headers)
        self.h2.send_data(stream_id, body, end_stream=True)
        self.sock.sendall(self.h2.data_to_send())
        status, chunks = None, []
        while True:
            data = self.sock.recv(65535)
            if not data:
                self.closed = True
                raise ConnectionError("APNs closed the connection")
            for event in self.h2.receive_data(data):
                if isinstance(event, h2.events.ResponseReceived) and event.stream_id == stream_id:
                    status = int(dict(event.headers)[":status"])
                elif isinstance(event, h2.events.DataReceived) and event.stream_id == stream_id:
                    chunks.append(event.data)
                    self.h2.acknowledge_received_data(event.flow_controlled_length, stream_id)
                elif isinstance(event, h2.events.StreamReset) and event.stream_id == stream_id:
                    raise ConnectionError("APNs reset the stream")
                elif isinstance(event, h2.events.ConnectionTerminated):
                    self.closed = True
                elif isinstance(event, h2.events.StreamEnded) and event.stream_id == stream_id:
                    pending = self.h2.data_to_send()
                    if pending:
                        self.sock.sendall(pending)
                    return status, b"".join(chunks)
            pending = self.h2.data_to_send()
            if pending:
                self.sock.sendall(pending)

    def close(self):
        self.closed = True
        try:
            self.sock.close()
        except OSError:
            pass


class APNsClient:
    """Synchronous APNs sender. One HTTP/2 connection per environment. Not thread-safe:
    only the dispatcher thread calls send()."""

    def __init__(self, signer, connect=tls_connect, clock=time.monotonic):
        self.signer = signer
        self._connect = connect
        self.clock = clock
        self._connections = {}

    def send(self, environment, token_hex, topic, collapse_id, payload):
        host = HOSTS[environment]
        headers = [(":method", "POST"), (":scheme", "https"), (":authority", host),
                   (":path", f"/3/device/{token_hex}"),
                   ("authorization", f"bearer {self.signer.token()}"),
                   ("apns-push-type", "alert"), ("apns-priority", "10"),
                   ("apns-topic", topic), ("apns-collapse-id", collapse_id),
                   ("content-type", "application/json"), ("content-length", str(len(payload)))]
        try:
            connection = self._connection(environment, host)
            status, body = connection.request(headers, payload)
            connection.last_used = self.clock()
        except Exception as e:
            RNS.log(f"APNs request to {host} failed: {e}", RNS.LOG_WARNING)
            self._drop(environment)
            return CONNECTION_FAILED
        if connection.closed:
            self._drop(environment)
        reason = None
        if status != 200:
            try:
                reason = json.loads(body.decode("utf-8")).get("reason")
            except Exception:
                reason = None
        return status, reason

    def invalidate_token(self):
        self.signer.invalidate()

    def close(self):
        for environment in list(self._connections):
            self._drop(environment)

    def _connection(self, environment, host):
        connection = self._connections.get(environment)
        if connection is not None and (connection.closed or self.clock() - connection.last_used >= IDLE_REOPEN):
            self._drop(environment)
            connection = None
        if connection is None:
            connection = _Connection(self._connect(host), self.clock())
            self._connections[environment] = connection
        return connection

    def _drop(self, environment):
        connection = self._connections.pop(environment, None)
        if connection is not None:
            connection.close()
```

```python
import base64
import collections
import json
import threading
import time

import RNS

WINDOW = 0.5
MAX_PACKETS = 8
MAX_B64 = 3500
RATE_LIMIT = 20
RATE_WINDOW = 60.0
QUEUE_CAP = 1024
RETRY_DELAY = 1.0
PUSH_TITLE = "New message"
UNREGISTERED_REASONS = ("BadDeviceToken", "DeviceTokenNotForTopic")


def build_payload(packets_b64):
    return json.dumps({"aps": {"alert": {"title": PUSH_TITLE}, "mutable-content": 1, "content-available": 1},
                       "rns": {"v": 1, "packets": list(packets_b64)}},
                      separators=(",", ":")).encode("utf-8")


class _Batch:
    def __init__(self, opened):
        self.opened = opened
        self.packets = []
        self.size = 0


class PushDispatcher:
    """Turns packets handed to dormant interfaces into APNs pushes. enqueue() never blocks;
    a single worker thread batches, rate-limits and sends."""

    def __init__(self, registry, apns, on_unregistered, clock=time.monotonic, sleep=time.sleep, start_thread=True):
        self.registry = registry
        self.apns = apns
        self.on_unregistered = on_unregistered
        self.clock = clock
        self.sleep = sleep
        self._queue = collections.deque()
        self._lock = threading.Lock()
        self._wake = threading.Event()
        self._stopped = False
        self._batches = {}
        self._history = collections.defaultdict(collections.deque)
        if start_thread:
            threading.Thread(target=self._run, daemon=True).start()

    def enqueue(self, destination_hash, raw):
        with self._lock:
            if len(self._queue) >= QUEUE_CAP:
                self._queue.popleft()
                RNS.log("Push queue is full, dropping the oldest packet", RNS.LOG_WARNING)
            self._queue.append((destination_hash, raw))
        self._wake.set()

    def step(self):
        """Move queued packets into batches and flush the ones that are due.
        Returns the seconds until the next batch is due, or None when nothing is pending."""
        with self._lock:
            items = list(self._queue)
            self._queue.clear()
        for destination_hash, raw in items:
            self._add(destination_hash, raw)
        now = self.clock()
        for destination_hash in [h for h, b in self._batches.items() if now - b.opened >= WINDOW]:
            self._flush(destination_hash)
        if not self._batches:
            return None
        return max(0.0, min(b.opened for b in self._batches.values()) + WINDOW - self.clock())

    def stop(self):
        self._stopped = True
        self._wake.set()

    def _run(self):
        timeout = None
        while not self._stopped:
            self._wake.wait(timeout)
            self._wake.clear()
            if self._stopped:
                break
            try:
                timeout = self.step()
            except Exception as e:
                RNS.log(f"Push dispatcher error: {e}", RNS.LOG_ERROR)
                timeout = WINDOW

    def _add(self, destination_hash, raw):
        entry = base64.b64encode(raw).decode("ascii")
        batch = self._batches.get(destination_hash)
        if batch is not None and batch.size + len(entry) > MAX_B64:
            self._flush(destination_hash)
            batch = None
        if batch is None:
            batch = _Batch(self.clock())
            self._batches[destination_hash] = batch
        batch.packets.append(entry)
        batch.size += len(entry)
        if len(batch.packets) >= MAX_PACKETS:
            self._flush(destination_hash)

    def _flush(self, destination_hash):
        batch = self._batches.pop(destination_hash)
        registration = self.registry.get(destination_hash)
        if registration is None:
            RNS.log(f"Dropping push for unregistered {RNS.prettyhexrep(destination_hash)}", RNS.LOG_DEBUG)
            return
        now = self.clock()
        history = self._history[destination_hash]
        while history and now - history[0] >= RATE_WINDOW:
            history.popleft()
        if len(history) >= RATE_LIMIT:
            RNS.log(f"Push rate limit reached for {RNS.prettyhexrep(destination_hash)}, dropping", RNS.LOG_DEBUG)
            return
        history.append(now)
        self._send(registration, build_payload(batch.packets))

    def _send(self, registration, payload):
        arguments = (registration.environment, registration.token.hex(), registration.bundle_id,
                     registration.destination_hash.hex(), payload)
        status, reason = self.apns.send(*arguments)
        if status == 0 or status == 429 or status >= 500:
            self.sleep(RETRY_DELAY)
            status, reason = self.apns.send(*arguments)
        elif status == 403 and reason == "ExpiredProviderToken":
            self.apns.invalidate_token()
            status, reason = self.apns.send(*arguments)
        if status == 200:
            return
        if status == 410 or (status == 400 and reason in UNREGISTERED_REASONS):
            RNS.log(f"APNs rejected the token for {RNS.prettyhexrep(registration.destination_hash)} ({status} {reason}), unregistering", RNS.LOG_NOTICE)
            self.on_unregistered(registration.destination_hash)
            return
        RNS.log(f"APNs push for {RNS.prettyhexrep(registration.destination_hash)} failed: {status} {reason}", RNS.LOG_ERROR)
```

```python
import socket

import RNS
from RNS.Interfaces.TCPInterface import TCPClientInterface
from RNS.Transport import IDX_RT_OUTB_IF

from .packet import parse_header

PUSHABLE_TYPES = (RNS.Packet.DATA, RNS.Packet.LINKREQUEST)


class PushClientInterface(TCPClientInterface):
    """A spawned hub connection. Once bound to a registration, it survives its socket:
    it stays attached to Transport, and what Transport hands it goes out as a push."""

    def __init__(self, owner, configuration, connected_socket=None, push_server=None):
        super().__init__(owner, configuration, connected_socket=connected_socket)
        self.push_server = push_server
        self.bound_hash = None
        self.dormant = False

    def teardown(self):
        if self.bound_hash is None:
            return super().teardown()
        self.online = False
        if not self.dormant:
            self.dormant = True
            RNS.log(f"{self} is dormant for {RNS.prettyhexrep(self.bound_hash)}", RNS.LOG_INFO)
        if self.socket is not None:
            try:
                self.socket.shutdown(socket.SHUT_RDWR)
            except OSError:
                pass
            try:
                self.socket.close()
            except OSError:
                pass

    def retire(self):
        RNS.log(f"Retiring {self}", RNS.LOG_VERBOSE)
        self.dormant = False
        self.bound_hash = None
        self.online = False
        self.OUT = False
        self.IN = False
        parent = self.parent_interface
        if parent is not None:
            while self in parent.spawned_interfaces:
                parent.spawned_interfaces.remove(self)
        RNS.Transport.remove_interface(self)

    def process_outgoing(self, data):
        if not self.dormant:
            return super().process_outgoing(data)
        try:
            header = parse_header(data)
            if header is not None and header.destination_hash == self.bound_hash and header.packet_type in PUSHABLE_TYPES:
                self.push_server.dispatcher.enqueue(header.destination_hash, data)
        except Exception as e:
            RNS.log(f"Could not queue a push from {self}: {e}", RNS.LOG_ERROR)

    def process_incoming(self, data):
        target = self._proof_target(data)
        if target is None:
            return super().process_incoming(data)
        if self.online and not self.detached:
            self.rxb += len(data)
            if self.parent_interface is not None:
                self.parent_interface.rxb += len(data)
            self.owner.inbound(data, target)

    def _proof_target(self, data):
        """Stock Transport forwards a proof only if it arrives on the interface the proved
        packet left through. When that interface is one of our dormant ones, the proof
        arrives on a fresh connection instead, so hand it to Transport as the dormant one's."""
        header = parse_header(data)
        if header is None or header.packet_type != RNS.Packet.PROOF:
            return None
        entry = RNS.Transport.reverse_table.get(header.destination_hash)
        if entry is None:
            return None
        outbound = entry[IDX_RT_OUTB_IF]
        if (isinstance(outbound, PushClientInterface) and outbound is not self and outbound.dormant
                and outbound.push_server is self.push_server):
            return outbound
        return None
```

```python
import collections
import os
import threading
import time

import RNS
from RNS.Transport import IDX_PT_RVCD_IF
from RNS.vendor import umsgpack

from .registry import ENVIRONMENTS, Registration, delivery_hash

APP_NAME = "rnspush"
ASPECT = "registration"
ANNOUNCE_INTERVAL = 30 * 60
SWEEP_INTERVAL = 60 * 60
REQUEST_LIMIT = 10
REQUEST_WINDOW = 60 * 60
MAX_TOKEN_LENGTH = 100


def _error(code):
    return {"ok": False, "error": code}


def find_active_link(link_id):
    for link in list(RNS.Transport.active_links):
        if link.link_id == link_id:
            return link
    return None


def path_interface(destination_hash):
    entry = RNS.Transport.path_table.get(destination_hash)
    return entry[IDX_PT_RVCD_IF] if entry is not None else None


def load_or_create_identity(path):
    if os.path.isfile(path):
        identity = RNS.Identity.from_file(path)
        if identity is None:
            raise ValueError(f"Could not load the push hub identity from {path}")
        return identity
    os.makedirs(os.path.dirname(path), exist_ok=True)
    identity = RNS.Identity()
    identity.to_file(path)
    return identity


class _BadRequest(Exception):
    pass


class RegistrationService:
    """The rnspush.registration destination: /register and /unregister over an identified
    link, plus an lxmf.delivery announce handler that re-binds reconnecting clients."""

    aspect_filter = "lxmf.delivery"

    def __init__(self, server, registry, allowed_bundle_ids, clock=time.time,
                 find_link=find_active_link, path_interface=path_interface):
        self.server = server
        self.registry = registry
        self.allowed_bundle_ids = frozenset(allowed_bundle_ids)
        self.clock = clock
        self.find_link = find_link
        self.path_interface = path_interface
        self.destination = None
        self._requests = collections.defaultdict(collections.deque)
        self._lock = threading.Lock()
        self._stopped = threading.Event()

    def start(self, identity):
        destination = RNS.Destination(identity, RNS.Destination.IN, RNS.Destination.SINGLE, APP_NAME, ASPECT)
        destination.register_request_handler("/register", response_generator=self.handle_register,
                                             allow=RNS.Destination.ALLOW_ALL)
        destination.register_request_handler("/unregister", response_generator=self.handle_unregister,
                                             allow=RNS.Destination.ALLOW_ALL)
        self.destination = destination
        RNS.Transport.register_announce_handler(self)
        RNS.log(f"Push registration destination is {RNS.prettyhexrep(destination.hash)}", RNS.LOG_NOTICE)
        threading.Thread(target=self._maintenance_loop, daemon=True).start()

    def stop(self):
        self._stopped.set()

    def announce(self):
        self.destination.announce(app_data=umsgpack.packb({"v": 1}))

    def sweep(self):
        for destination_hash in self.registry.expire():
            self.server.retire(destination_hash)

    def handle_register(self, path, data, request_id, link_id, remote_identity, requested_at):
        try:
            interface, error = self._preflight(link_id, remote_identity)
            if error is not None:
                return error
            token, environment, bundle_id = self._parse_register(data)
            if bundle_id not in self.allowed_bundle_ids:
                return _error("bundle_not_allowed")
            destination_hash = delivery_hash(remote_identity)
            registration = Registration(destination_hash, remote_identity.get_public_key(), token,
                                        environment, bundle_id, self.clock())
            self.registry.put(registration)
            self.server.bind(destination_hash, interface)
            RNS.log(f"Registered {RNS.prettyhexrep(destination_hash)} for push", RNS.LOG_INFO)
            return {"ok": True, "destination_hash": destination_hash, "expires_at": int(registration.expires_at)}
        except _BadRequest:
            return _error("bad_request")
        except Exception as e:
            RNS.log(f"Push /register failed: {e}", RNS.LOG_ERROR)
            return _error("bad_request")

    def handle_unregister(self, path, data, request_id, link_id, remote_identity, requested_at):
        try:
            _, error = self._preflight(link_id, remote_identity)
            if error is not None:
                return error
            if not isinstance(data, dict) or data.get("v") != 1:
                return _error("bad_request")
            destination_hash = delivery_hash(remote_identity)
            self.registry.remove(destination_hash)
            self.server.retire(destination_hash)
            return {"ok": True}
        except Exception as e:
            RNS.log(f"Push /unregister failed: {e}", RNS.LOG_ERROR)
            return _error("bad_request")

    def received_announce(self, destination_hash, announced_identity, app_data):
        try:
            registration = self.registry.get(destination_hash)
            if registration is None or announced_identity is None:
                return
            if announced_identity.get_public_key() != registration.identity_public_key:
                return
            interface = self.path_interface(destination_hash)
            if self.server.owns(interface):
                self.server.bind(destination_hash, interface)
        except Exception as e:
            RNS.log(f"Push announce handler failed: {e}", RNS.LOG_ERROR)

    def _preflight(self, link_id, remote_identity):
        if remote_identity is None:
            return None, _error("not_identified")
        if not self._admit(remote_identity.hash):
            return None, _error("rate_limited")
        link = self.find_link(link_id)
        interface = getattr(link, "attached_interface", None) if link is not None else None
        if not self.server.owns(interface):
            return None, _error("not_direct")
        return interface, None

    def _parse_register(self, data):
        if not isinstance(data, dict) or data.get("v") != 1:
            raise _BadRequest()
        token = data.get("token")
        environment = data.get("environment")
        bundle_id = data.get("bundle_id")
        if not isinstance(token, bytes) or not 1 <= len(token) <= MAX_TOKEN_LENGTH:
            raise _BadRequest()
        if not isinstance(environment, str) or environment not in ENVIRONMENTS:
            raise _BadRequest()
        if not isinstance(bundle_id, str) or not bundle_id:
            raise _BadRequest()
        return token, environment, bundle_id

    def _admit(self, identity_hash):
        now = self.clock()
        with self._lock:
            history = self._requests[identity_hash]
            while history and now - history[0] >= REQUEST_WINDOW:
                history.popleft()
            if len(history) >= REQUEST_LIMIT:
                return False
            history.append(now)
            return True

    def _maintenance_loop(self):
        last_announce = None
        last_sweep = time.monotonic()
        while not self._stopped.is_set():
            now = time.monotonic()
            if last_announce is None or now - last_announce >= ANNOUNCE_INTERVAL:
                try:
                    self.announce()
                except Exception as e:
                    RNS.log(f"Could not announce the push registration destination: {e}", RNS.LOG_ERROR)
                last_announce = now
            if now - last_sweep >= SWEEP_INTERVAL:
                try:
                    self.sweep()
                except Exception as e:
                    RNS.log(f"Push registration sweep failed: {e}", RNS.LOG_ERROR)
                last_sweep = now
            self._stopped.wait(10)
```

```python
def default_configdir():
    """The same search order rnsd uses."""
    if os.path.isfile("/etc/reticulum/config"):

    user_config = os.path.expanduser("~/.config/reticulum")

def _live_registrations(path):
    try:
        with open(path) as f:
            entries = json.load(f)["registrations"]
    except (OSError, ValueError, KeyError):
        return 0
    now = time.time()
    return sum(1 for entry in entries if now < float(entry["registered_at"]) + REGISTRATION_LIFETIME)

    status = commands.add_parser("status", help="show the registration destination and registration count")
```

````markdown
# rns-push

A push-capable hub for [Reticulum](https://reticulum.network). It runs inside stock `rnsd` as an
external interface plugin. When a registered iOS client's TCP connection drops, the hub keeps

in [docs/server-contract.md](docs/server-contract.md).

pip install rns-push

## Apple setup

In the Apple Developer portal, create an APNs authentication key (Keys → +, enable Apple Push
Notifications service). Download the `.p8` file once, and note its Key ID and your Team ID. The
app's bundle id must belong to that team. No server-side domain verification is needed.

In the Reticulum `config`:

```
[reticulum]
  enable_transport = Yes

[interfaces]
  [[Push Hub]]
    type = PushTCPServerInterface
    enabled = yes
    listen_ip = 0.0.0.0
    listen_port = 4242
    apns_key_file = /etc/rns-push/AuthKey_ABC123.p8
    apns_key_id = ABC123
    apns_team_id = TEAM123456

```

All `TCPServerInterface` options work except IFAC (`network_name`, `passphrase`, `ifac_size`),
which this interface refuses. Transport must be enabled.

Start `rnsd`, then run `rns-push status` to print the registration destination hash. Clients
enter that hash as their push hub destination. State lives in `<config>/storage/rns_push/`.

- Dormant interfaces do not survive an `rnsd` restart. Pushing resumes the next time each
  client connects and announces.
- The hub forgets a client's path after a week without a fresh announce (Reticulum's

- Notifications always read "New message". The hub never sees message content.
````

```python
import base64
import json
import os
import shutil
import socket
import subprocess
import sys
import tempfile
import time
import unittest

import RNS
from RNS.Interfaces.TCPInterface import HDLC

from rns_push.registry import delivery_hash
from tests.support import p256_pem

TESTS_DIR = os.path.dirname(os.path.abspath(__file__))
REPO = os.path.dirname(TESTS_DIR)
E2E = os.path.join(TESTS_DIR, "e2e")

HUB_CONFIG = """[reticulum]
  enable_transport = Yes
  share_instance = No
  panic_on_interface_error = No

[logging]
  loglevel = 5

[interfaces]
  [[Push Hub]]
    type = PushTCPServerInterface
    enabled = yes
    listen_ip = 127.0.0.1
    listen_port = {port}
    apns_key_file = {key}
    apns_key_id = KEY1234567
    apns_team_id = TEAM123456
    allowed_bundle_ids = co.example.push
"""

CLIENT_CONFIG = """[reticulum]
  enable_transport = No
  share_instance = No

[logging]
  loglevel = 5

[interfaces]
  [[Hub]]
    type = TCPClientInterface
    enabled = yes
    target_host = 127.0.0.1
    target_port = {port}
"""


def free_port():
    with socket.socket() as s:
        s.bind(("127.0.0.1", 0))
        return s.getsockname()[1]


def wait_for_file(path, timeout):
    deadline = time.time() + timeout
    while not os.path.isfile(path) or os.path.getsize(path) == 0:
        if time.time() > deadline:
            raise AssertionError(f"timed out waiting for {path}")
        time.sleep(0.1)
    with open(path) as f:
        return f.read()


def implicit_proof(raw, identity):
    packet = RNS.Packet(None, raw)
    packet.unpack()
    flags = (RNS.Packet.HEADER_1 << 6) | (RNS.Transport.BROADCAST << 4) | (RNS.Destination.SINGLE << 2) | RNS.Packet.PROOF
    return bytes([flags, 0]) + packet.packet_hash[:16] + bytes([RNS.Packet.NONE]) + identity.sign(packet.packet_hash)


class EndToEndTest(unittest.TestCase):
    def setUp(self):
        self.dir = tempfile.mkdtemp()
        self.processes = []
        self.port = free_port()
        _, pem = p256_pem()
        key_path = os.path.join(self.dir, "AuthKey.p8")
        with open(key_path, "wb") as f:
            f.write(pem)
        self.hub_dir = self.config_dir("hub", HUB_CONFIG.format(port=self.port, key=key_path))
        os.makedirs(os.path.join(self.hub_dir, "interfaces"))
        shutil.copy(os.path.join(REPO, "shim", "PushTCPServerInterface.py"), os.path.join(self.hub_dir, "interfaces"))
        self.client_dir = self.config_dir("client", CLIENT_CONFIG.format(port=self.port))
        self.sender_dir = self.config_dir("sender", CLIENT_CONFIG.format(port=self.port))
        self.identity = RNS.Identity()
        self.identity_path = os.path.join(self.dir, "client_identity")
        self.identity.to_file(self.identity_path)

    def tearDown(self):
        for process in self.processes:
            process.kill()
            process.wait()
        shutil.rmtree(self.dir)

    def config_dir(self, name, config):
        path = os.path.join(self.dir, name)
        os.makedirs(path)
        with open(os.path.join(path, "config"), "w") as f:
            f.write(config)
        return path

    def spawn(self, name, *args):
        log = open(os.path.join(self.dir, f"{name}.log"), "w")
        self.addCleanup(log.close)
        process = subprocess.Popen([sys.executable, os.path.join(E2E, f"{name}.py"), *args],
                                   stdout=log, stderr=subprocess.STDOUT)
        self.processes.append(process)
        return process

    def path(self, name):
        return os.path.join(self.dir, name)

    def test_a_pushed_packet_is_proved_back_to_the_sender(self):
        self.spawn("hub", self.hub_dir, self.path("pushes.jsonl"), self.path("ready.json"))
        registration_hex = json.loads(wait_for_file(self.path("ready.json"), 30))["registration_hash"]

        client = self.spawn("client", self.client_dir, self.identity_path, registration_hex, self.path("registered.json"))
        self.assertEqual(client.wait(timeout=60), 0, "client failed, see client.log")
        registered = json.loads(wait_for_file(self.path("registered.json"), 5))
        client_hex = delivery_hash(self.identity).hex()
        self.assertEqual(registered, {"ok": True, "error": None, "destination_hash": client_hex})
        time.sleep(1.0)  # let the hub see the closed socket and go dormant

        self.spawn("sender", self.sender_dir, client_hex, self.path("sent.json"), self.path("receipt.json"))
        wait_for_file(self.path("sent.json"), 40)
        push = json.loads(wait_for_file(self.path("pushes.jsonl"), 20).splitlines()[0])
        self.assertEqual(push["environment"], "sandbox")
        self.assertEqual(push["token"], "ab" * 32)
        self.assertEqual(push["topic"], "co.example.push")
        self.assertEqual(push["collapse_id"], client_hex)
        payload = json.loads(push["payload"])
        self.assertEqual(payload["aps"], {"alert": {"title": "New message"}, "mutable-content": 1, "content-available": 1})
        raw = base64.b64decode(payload["rns"]["packets"][0])

        frame = bytes([HDLC.FLAG]) + HDLC.escape(implicit_proof(raw, self.identity)) + bytes([HDLC.FLAG])
        with socket.create_connection(("127.0.0.1", self.port)) as extension:
            extension.sendall(frame)
            time.sleep(1.0)

        receipt = json.loads(wait_for_file(self.path("receipt.json"), 30))
        self.assertEqual(receipt["status"], RNS.PacketReceipt.DELIVERED, "see hub.log and sender.log")


if __name__ == "__main__":
    unittest.main()
```
